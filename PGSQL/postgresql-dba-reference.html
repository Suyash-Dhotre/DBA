Logical Backup — pg_dump
# Single DB — custom format (recommended)
pg_dump -Fc -d mydb -f mydb.dump

# Parallel dump — directory format, 4 jobs
pg_dump -Fd -j 4 -d mydb -f /backup/mydb_dir/

# All databases + globals
pg_dumpall -f full.sql
pg_dumpall -g -f globals_only.sql  # roles only

# Restore
pg_restore -d mydb -Fc --jobs=4 mydb.dump
pg_restore --list mydb.dump        # verify contents
Physical Backup — pg_basebackup
# Full backup with WAL streaming
pg_basebackup -h localhost -U replicator \
  -D /backup/base/ -Ft -z \
  --wal-method=stream -P

# Incremental (PostgreSQL 17+)
pg_basebackup -D /backup/inc1/ \
  --incremental=/backup/base/backup_manifest

# Verify backup integrity (PG13+)
pg_verifybackup /backup/base/

WAL Archiving & PITR

WAL Archiving Setup
# postgresql.conf
archive_mode = on
archive_command = 'cp %p /mnt/wal_archive/%f'
archive_timeout = 300   # force switch every 5 min

# Force WAL switch manually
SELECT pg_switch_wal();
SELECT * FROM pg_stat_archiver;
PITR Recovery Configuration
# postgresql.conf (recovery server)
restore_command = 'cp /mnt/wal_archive/%f %p'
recovery_target_time = '2026-05-27 10:30:00'
recovery_target_action = 'promote'

# Create recovery trigger
touch /pgdata/16/data/recovery.signal
# Start PG — recovers to target then promotes

Barman Backup Manager

✅ FREE & open source (EDB). Features: integrated PITR, native retention policies, WAL streaming, S3/Azure/GCS, Nagios plugin (check_barman).
barman.conf Configuration
[barman]
barman_home = /var/lib/barman
compression = gzip
parallel_jobs = 2

[pg-server]
description = "Primary PostgreSQL"
conninfo = host=pghost user=barman dbname=postgres
streaming_conninfo = host=pghost user=streaming_barman
backup_method = postgres
streaming_archiver = on
slot_name = barman_slot
retention_policy = RECOVERY WINDOW OF 14 DAYS
minimum_redundancy = 1
wal_retention_policy = main
Key Barman Commands
# Health check
barman check pg-server

# Take full backup
barman backup pg-server

# List backups
barman list-backup pg-server

# PITR restore (one command — fully integrated)
barman recover pg-server latest /pgdata \
  --target-time "2026-05-27 10:30:00" \
  --remote-ssh-command "ssh postgres@dbserver"

# WAL management & retention enforcement
barman switch-wal pg-server
barman cron
barman check-backup pg-server 20260527T100000

Memory Architecture

Memory Layout Diagram
SHARED MEMORY (OS shared memory segment) ├── Shared Buffers ← Main page cache (heap + index pages) │ ├── Buffer descriptors (metadata array) │ ├── 8kB data pages (actual page content) │ ├── Clock-sweep replacement algorithm │ └── Recommended: ~25% of total RAM ├── WAL Buffers ← wal_buffers (auto = 1/32 of shared_buffers) ├── Lock Space ← max_locks_per_transaction ├── Commit Timestamps ← track_commit_timestamp └── Background Workers ← max_worker_processes slots PER-PROCESS LOCAL MEMORY (one set per backend process) ├── work_mem ← Per sort/hash NODE per query (4MB default) ├── temp_buffers ← Per-session temp table cache (8MB) └── maintenance_work_mem ← VACUUM / CREATE INDEX (64MB default)
Shared Buffers Parameters

Total Memory Sizing Formula
# Total RAM estimate
  shared_buffers          (25% RAM)
+ max_connections
    × work_mem × avg_sort_nodes
+ max_connections × temp_buffers
+ maintenance_work_mem
    × autovacuum_max_workers
+ OS page cache           (~50% — PG relies on it)
+ wal_buffers

# Buffer hit ratio monitoring
SELECT sum(blks_hit) * 100.0 /
  (sum(blks_hit) + sum(blks_read)) AS hit_ratio
FROM pg_stat_database;

# Buffer contents (requires pg_buffercache ext)
SELECT c.relname, count(*) AS buffers
FROM pg_buffercache b
JOIN pg_class c ON b.relfilenode = pg_relation_filenode(c.oid)
GROUP BY c.relname ORDER BY buffers DESC LIMIT 20;
ℹ️ PostgreSQL double-buffering: a page can exist in both shared_buffers AND OS page cache. Setting shared_buffers to ~25% RAM accounts for this overlap. Enable huge_pages=on to reduce TLB pressure on large servers. Unlike MySQL InnoDB, PostgreSQL intentionally leverages the OS page cache.

Background Processes

Process Tree (Multi-Process Architecture)
postmaster (PID 1 — supervisor, listener, crash recovery coordinator) ├── checkpointer ← flushes dirty shared_buffers pages at checkpoint ├── background writer ← proactively writes dirty pages to reduce stalls ├── WAL writer ← periodically flushes WAL buffers → WAL segments ├── WAL archiver ← copies completed WAL segments to archive_command ├── autovacuum launcher ← schedules VACUUM/ANALYZE workers │ └── autovacuum worker (×autovacuum_max_workers, spawned dynamically) ├── stats collector ← pg_stat_* data collection (PG14 and below) ├── logical replication launcher │ └── logical replication worker (one per active subscription) ├── WAL receiver ← STANDBY ONLY: receives WAL stream from primary ├── startup process ← STANDBY ONLY: replays WAL (MRP equivalent) └── backend process ← one OS process per active client connection

Process Health Monitoring
-- Checkpoint health (watch buffers_backend — should be near 0)
SELECT checkpoints_timed, checkpoints_req,
       buffers_checkpoint, buffers_clean, buffers_backend
FROM pg_stat_bgwriter;

-- Autovacuum lag (top bloated tables)
SELECT schemaname, relname, n_dead_tup, last_autovacuum,
       n_dead_tup::float / NULLIF(n_live_tup,0) AS dead_ratio
FROM pg_stat_user_tables
ORDER BY n_dead_tup DESC LIMIT 10;

-- XID wraparound risk check
SELECT relname, age(relfrozenxid) AS xid_age
FROM pg_class WHERE relkind = 'r'
ORDER BY xid_age DESC LIMIT 10;
-- Alert if age > 150,000,000

-- WAL archiver health
SELECT * FROM pg_stat_archiver;
-- failed_count should be 0; last_failed_wal should be NULL

Installation RPM / DEB

RHEL / Oracle Linux / Rocky Linux
# Disable distro PostgreSQL module
dnf -qy module disable postgresql

# Add official PGDG repository
dnf install -y https://download.postgresql.org/pub/repos/\
yum/reporpms/EL-8-x86_64/pgdg-redhat-repo-latest.noarch.rpm

# Install server + client
dnf install -y postgresql16-server postgresql16

# Initialize data cluster
/usr/pgsql-16/bin/postgresql-16-setup initdb

# Enable and start
systemctl enable postgresql-16
systemctl start postgresql-16

# Verify
psql -V
systemctl status postgresql-16
Ubuntu / Debian
# Add PGDG apt repository
apt install -y curl ca-certificates
install -d /usr/share/postgresql-common/pgdg
curl -o /usr/share/postgresql-common/pgdg/apt.postgresql.org.asc \
  https://www.postgresql.org/media/keys/ACCC4CF8.asc

sh -c 'echo "deb [signed-by=...] \
  https://apt.postgresql.org/pub/repos/apt \
  $(lsb_release -cs)-pgdg main" \
  > /etc/apt/sources.list.d/pgdg.list'

# Install
apt update && apt install -y postgresql-16

# Ubuntu auto-initializes — just verify
systemctl status postgresql
psql -V
TAR Binary Installation
# Download and extract
wget https://get.enterprisedb.com/postgresql/\
postgresql-16.4-1-linux-x64-binaries.tar.gz
tar -xzf postgresql-16.4-1-linux-x64-binaries.tar.gz -C /opt/
mv /opt/pgsql /opt/pgsql-16.4
ln -s /opt/pgsql-16.4 /opt/pgsql      # symlink — key for upgrades

# Create OS user and directories
useradd -m -s /bin/bash postgres
mkdir -p /pgdata/16/{data,wal,archive} /pglog/16
chown -R postgres:postgres /pgdata /pglog /opt/pgsql-16.4
chmod 700 /pgdata/16/data

# Environment (~/.bash_profile for postgres user)
export PG_HOME=/opt/pgsql
export PGDATA=/pgdata/16/data
export PGPORT=5432
export PATH=$PG_HOME/bin:$PATH

# Initialize cluster with checksums
initdb -D /pgdata/16/data -E UTF8 --locale=en_US.UTF-8 -k

# Start
pg_ctl -D /pgdata/16/data start -l /pglog/16/startup.log

Minor & Major Upgrades

Minor Upgrade (e.g. 16.1 → 16.4)
# RPM-based — binary replace + restart (no initdb needed)
dnf update postgresql16-server postgresql16 -y
systemctl restart postgresql-16
psql -c "SELECT version();"

# TAR-based — update symlink, no data directory changes
pg_ctl -D /pgdata/16/data stop -m fast
tar -xzf postgresql-16.4-1-linux-x64-binaries.tar.gz -C /opt/
mv /opt/pgsql /opt/pgsql-16.4
ln -sfn /opt/pgsql-16.4 /opt/pgsql   # atomic symlink swap
pg_ctl -D /pgdata/16/data start -l /pglog/16/startup.log
psql -c "SELECT version();"
Major Upgrade — pg_upgrade (PG15 → PG16)
# Step 1: Install PG16, initialize new cluster
dnf install -y postgresql16-server postgresql16
/usr/pgsql-16/bin/postgresql-16-setup initdb

# Step 2: Stop old cluster
systemctl stop postgresql-15

# Step 3: Dry run — resolve ALL issues first
/usr/pgsql-16/bin/pg_upgrade \
  -b /usr/pgsql-15/bin  -B /usr/pgsql-16/bin \
  -d /var/lib/pgsql/15/data \
  -D /var/lib/pgsql/16/data \
  --check

# Step 4: Run upgrade
/usr/pgsql-16/bin/pg_upgrade \
  -b /usr/pgsql-15/bin  -B /usr/pgsql-16/bin \
  -d /var/lib/pgsql/15/data \
  -D /var/lib/pgsql/16/data \
  --link              # hard-link = instant; omit for copy mode
  -j 4                # parallel jobs (PG14+)

# Step 5: Start new cluster + update statistics
systemctl start postgresql-16
./analyze_new_cluster.sh    # mandatory post-upgrade
./delete_old_cluster.sh     # after confirming success
MethodDowntimeComplexityRollbackBest For
pg_upgrade --linkMinutesMediumHardLarge DBs, same server
pg_upgrade (copy)HoursMediumEasyMedium DBs, rollback safety
Dump & RestoreHours–DaysLowEasySmall DBs, cross-server
Logical ReplicationMinutes (cutover)HighEasyZero downtime production

Replication Setup

Streaming Replication Architecture
Primary Server ├── WAL Writer → pg_wal/ (segment files) └── WAL Sender process (one per standby) │ TCP streaming (replication protocol) ▼ Standby Server ├── WAL Receiver ← receives WAL, writes to pg_wal/ └── Startup Process ← replays WAL continuously (MRP equivalent) (hot_standby=on → allows SELECT queries while in recovery)
Primary Configuration
# postgresql.conf
wal_level = replica
max_wal_senders = 10
max_replication_slots = 10
wal_keep_size = 1GB
hot_standby = on
hot_standby_feedback = on
wal_log_hints = on       # required for pg_rewind

# pg_hba.conf
host replication replicator 10.0.0.0/8 scram-sha-256

# Create replication user
CREATE USER replicator REPLICATION LOGIN
  PASSWORD 'RepStrongPass123';

# Create physical replication slot
SELECT pg_create_physical_replication_slot('standby1_slot');
Standby Setup & Monitoring
# Take base backup — auto-creates standby.signal
pg_basebackup -h 10.0.0.1 -U replicator \
  -D /pgdata/16/data -Fp -Xs -P -R \
  -S standby1_slot   # attach to replication slot

# Start standby
pg_ctl -D /pgdata/16/data start

-- Monitor on PRIMARY
SELECT client_addr, state, sent_lsn, replay_lsn,
  pg_wal_lsn_diff(sent_lsn, replay_lsn) AS lag_bytes,
  sync_state
FROM pg_stat_replication;

-- Replication lag on STANDBY (seconds)
SELECT now() - pg_last_xact_replay_timestamp() AS lag;

-- Promote standby
SELECT pg_promote();
-- or: pg_ctl -D /pgdata promote
Synchronous & Cascading Replication
# Synchronous replication (postgresql.conf on PRIMARY)
synchronous_standby_names = 'FIRST 1 (standby1, standby2)'
synchronous_commit = remote_apply  # strongest guarantee

# synchronous_commit options:
#   off          → async (fastest, risk data loss)
#   local        → wait for local WAL flush only
#   on           → wait for standby WAL flush
#   remote_write → wait for standby OS write (not fsync)
#   remote_apply → wait for standby to replay (strongest)

# Delayed standby (recovery_min_apply_delay)
recovery_min_apply_delay = '30min'  # on standby postgresql.conf
# Useful for: accidental DROP TABLE recovery window

# Cascading: Primary → Standby1 → Standby2
# On Standby2: primary_conninfo points to Standby1
# Standby1 must have: max_wal_senders=5, wal_level=replica
Logical Replication (pub/sub) — PG10+
-- Requires: wal_level = logical (STATIC — restart needed)

-- ON PUBLISHER
CREATE PUBLICATION mypub FOR TABLE orders, customers;
-- or replicate all tables:
CREATE PUBLICATION mypub_all FOR ALL TABLES;

-- ON SUBSCRIBER (schema must already exist — DDL not replicated)
CREATE SUBSCRIPTION mysub
  CONNECTION 'host=10.0.0.1 port=5432 dbname=mydb
              user=replicator password=RepStrongPass123'
  PUBLICATION mypub;

-- Monitor sync status
SELECT subname, pid, received_lsn, latest_end_lsn,
       last_msg_receipt_time
FROM pg_stat_subscription;

SELECT * FROM pg_subscription_rel;
-- srsubstate: i=init, d=data copy, s=synchronized, r=streaming
⚠️ Logical replication does NOT replicate DDL (ALTER TABLE, CREATE INDEX) or sequences. Schema changes must be applied manually on subscriber.
