== Create user
CREATE ROLE repluser WITH REPLICATION LOGIN PASSWORD 'StrongPass';
==


--> Publisher
CREATE DATABASE salesdb;

\c salesdb

CREATE TABLE customers(
 id int PRIMARY KEY,
 name text,
 city text
);

INSERT INTO customers VALUES
(1,'Suyash','Nashik'),
(2,'Rahul','Pune');

CREATE PUBLICATION pub_sales FOR TABLE customers;


grant all on customers to repluser ;
grant all on salesdb to repluser ;



--> Subscriber
CREATE DATABASE salesdb;

\c salesdb
 
CREATE TABLE customers(
 id int PRIMARY KEY,
 name text,
 city text
);

CREATE SUBSCRIPTION sub_sales
CONNECTION 'host=192.168.56.101 port=5432 dbname=salesdb user=repluser password=123'
PUBLICATION pub_sales;


** Remove Replication
--> Subscriber
DROP SUBSCRIPTION sub_sales;

--> Publisher
DROP PUBLICATION pub_sales;
