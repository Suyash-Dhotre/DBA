== Replication Lag
SELECT pg_size_pretty(
pg_wal_lsn_diff(sent_lsn,replay_lsn)
)
FROM pg_stat_replication;

SELECT application_name,
       state,
       sync_state,
       pg_size_pretty(
         pg_wal_lsn_diff(sent_lsn,replay_lsn)
       ) AS lag
FROM pg_stat_replication;
