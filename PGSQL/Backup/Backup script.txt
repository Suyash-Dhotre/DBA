#!/bin/bash

#########################################
# PostgreSQL Daily Physical Backup Script
#########################################

# Variables
DATE=$(date +"%Y-%m-%d_%H-%M-%S")
BACKUP_BASE="/backup/postgres"
BACKUP_DIR="$BACKUP_BASE/$DATE"
LOG_DIR="$BACKUP_BASE/logs"
LOG_FILE="$LOG_DIR/backup_$DATE.log"

PGUSER="postgres"
PGHOST="localhost"
RETENTION_DAYS=7

#########################################
# Create Directories
#########################################

mkdir -p "$BACKUP_DIR"
mkdir -p "$LOG_DIR"

echo "========================================" >> $LOG_FILE
echo "Backup Started : $(date)" >> $LOG_FILE
echo "========================================" >> $LOG_FILE

#########################################
# Take pg_basebackup
#########################################

pg_basebackup \
-U $PGUSER \
-h $PGHOST \
-D "$BACKUP_DIR" \
-F tar \
-z \
-P \
-v \
-X stream >> $LOG_FILE 2>&1

BACKUP_STATUS=$?

#########################################
# Check Backup Status
#########################################

if [ $BACKUP_STATUS -eq 0 ]
then
    echo "" >> $LOG_FILE
    echo "Backup SUCCESS : $(date)" >> $LOG_FILE
else
    echo "" >> $LOG_FILE
    echo "Backup FAILED : $(date)" >> $LOG_FILE
    exit 1
fi

#########################################
# Remove Old Backups
#########################################

find $BACKUP_BASE -maxdepth 1 -type d -mtime +$RETENTION_DAYS ! -name logs -exec rm -rf {} \; >> $LOG_FILE 2>&1

#########################################
# Remove Old Logs
#########################################

find $LOG_DIR -type f -mtime +30 -exec rm -f {} \; >> $LOG_FILE 2>&1

echo "" >> $LOG_FILE
echo "Old Backup Cleanup Completed" >> $LOG_FILE

echo "========================================" >> $LOG_FILE
echo "Backup Finished : $(date)" >> $LOG_FILE
echo "========================================" >> $LOG_FILE

#!/bin/bash

#############################################
# PostgreSQL Full + Incremental Backup Script
#############################################

DATE=$(date +"%Y-%m-%d_%H-%M-%S")

BASE_DIR="/backup/postgres"
FULL_DIR="$BASE_DIR/full"
INCR_DIR="$BASE_DIR/incr"
LOG_DIR="$BASE_DIR/logs"

LOG_FILE="$LOG_DIR/backup_$DATE.log"

PGUSER="postgres"
PGHOST="localhost"

RETENTION_DAYS=7

DAY=$(date +%u)   # 1=Mon ... 7=Sun

mkdir -p "$FULL_DIR" "$INCR_DIR" "$LOG_DIR"

echo "========================================" >> $LOG_FILE
echo "Backup Start: $(date)" >> $LOG_FILE
echo "========================================" >> $LOG_FILE

#############################################
# FULL BACKUP (SUNDAY)
#############################################

if [ "$DAY" -eq 7 ]; then

    echo "Running FULL backup..." >> $LOG_FILE

    BACKUP_PATH="$FULL_DIR/full_$DATE"

    mkdir -p "$BACKUP_PATH"

    pg_basebackup \
        -U $PGUSER \
        -h $PGHOST \
        -D "$BACKUP_PATH" \
        -F tar \
        -z \
        -P \
        -v \
        -X stream \
        >> $LOG_FILE 2>&1

    if [ $? -eq 0 ]; then
        echo "FULL backup SUCCESS" >> $LOG_FILE
        # create manifest reference file marker
        touch "$BACKUP_PATH/FULL_OK"
    else
        echo "FULL backup FAILED" >> $LOG_FILE
        exit 1
    fi

#############################################
# INCREMENTAL BACKUP (MON-SAT)
#############################################

else

    echo "Running INCREMENTAL backup..." >> $LOG_FILE

    # Get latest full backup manifest
    LATEST_FULL=$(ls -1td $FULL_DIR/* 2>/dev/null | head -1)

    if [ -z "$LATEST_FULL" ]; then
        echo "No FULL backup found. Exiting." >> $LOG_FILE
        exit 1
    fi

    MANIFEST="$LATEST_FULL/backup_manifest"

    if [ ! -f "$MANIFEST" ]; then
        echo "Manifest not found in FULL backup. Exiting." >> $LOG_FILE
        exit 1
    fi

    INCR_PATH="$INCR_DIR/incr_$DATE"
    mkdir -p "$INCR_PATH"

    pg_basebackup \
        -U $PGUSER \
        -h $PGHOST \
        --incremental="$MANIFEST" \
        -D "$INCR_PATH" \
        -F tar \
        -z \
        -P \
        -v \
        -X stream \
        >> $LOG_FILE 2>&1

    if [ $? -eq 0 ]; then
        echo "INCREMENTAL backup SUCCESS" >> $LOG_FILE
        touch "$INCR_PATH/INCR_OK"
    else
        echo "INCREMENTAL backup FAILED" >> $LOG_FILE
        exit 1
    fi

fi

#############################################
# CLEANUP OLD BACKUPS
#############################################

find $BASE_DIR -type d -mtime +$RETENTION_DAYS -exec rm -rf {} \; >> $LOG_FILE 2>&1

#############################################
# CLEANUP OLD LOGS
#############################################

find $LOG_DIR -type f -mtime +30 -exec rm -f {} \; >> $LOG_FILE 2>&1

echo "Backup Completed: $(date)" >> $LOG_FILE
echo "========================================" >> $LOG_FILE
