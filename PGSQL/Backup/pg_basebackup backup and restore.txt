###== With tablespace outside Data Directory
== Plan restoration
pg_basebackup -h localhost -U postgres -D /backup/bkp -Fp -Xs -P -v \
-T /var/lib/pgsql/17/app_tablespc=/backup/bkp/app_tablespc 

pg_ctl start -D /var/lib/pgsql/17/backups/backup_restore


== Tar restoration
pg_basebackup -h localhost -U postgres -D /backup/bkp/ -Ft -Xs -P -v 

cd /backup/bkp/
tar -xvf base.tar -C /backup/bkp/

cat tablespace_map.old
32889 /var/lib/pgsql/17/app_tablespc

mv /var/lib/pgsql/17/app_tablespc/* /backup/archive_old

tar -xvf 32889.tar -C /var/lib/pgsql/17/app_tablespc

cd /backup/bkp/pg_tblspc
ln -s /var/lib/pgsql/17/app_tablespc /backup/bkp/pg_tblspc/32889
chmod 0700 -R /backup/bkp/

cd /backup/bkp/
mv pg_wal pg_wal_old
tar -xf pg_wal.tar -C /backup/bkp/pg_wal


pg_ctl start -D /backup/bkp/
